cmake_minimum_required(VERSION 3.20)
project(webWakeLock LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "Use emcmake for this Web build/link and Node probe")
endif()
option(WAKE_LOCK_PTHREADS "Exercise the shared main-loop request mechanism" OFF)
if(WAKE_LOCK_PTHREADS)
    add_compile_options(-pthread)
    add_link_options(-pthread -sPTHREAD_POOL_SIZE=1)
endif()
add_subdirectory(../../.. trussc)
add_executable(webWakeLock main.cpp)
target_compile_options(webWakeLock PRIVATE -UNDEBUG)
target_link_libraries(webWakeLock PRIVATE tc::TrussC)
target_link_options(webWakeLock PRIVATE -sENVIRONMENT=node -sFETCH=1
    -sALLOW_MEMORY_GROWTH=1 --pre-js=${CMAKE_CURRENT_SOURCE_DIR}/stub.js)
if(TC_WEB_BACKEND STREQUAL "GLES3")
    target_link_options(webWakeLock PRIVATE -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2)
else()
    target_link_options(webWakeLock PRIVATE --use-port=emdawnwebgpu)
endif()
