# core/tests/hotReloadScan — standalone regression test for the TC_HOT_RELOAD
# source scan (#234).
#
# The configure step and the pre-build check both decide "does this project use
# hot reload" through tc_hot_reload_scan() (core/cmake/tc_hot_reload_scan.cmake).
# When they used different file sets and comment rules, a project with its App
# in a src/ subdirectory could never build. This test feeds the shared rule the
# layouts and comment forms that used to split the two, at configure time, and
# compiles the verdicts into a tiny C program that reports them (exit code =
# failures). It also covers the platform gate in tc_hot_reload_decide() (#329):
# on platforms without hot reload (web / Android / iOS) the macro must decide
# OFF, in the configure step and in the pre-build check alike. Plain CMake, no TrussC: build_all.py runs it under
# --core-tests-only because of this committed CMakeLists.txt.
cmake_minimum_required(VERSION 3.16)
project(hotReloadScan C)

include(${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/tc_hot_reload_scan.cmake)

set(FX "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
set(_cases "")
set(_code "")

# case(<name> <expected ON|OFF> <files>...)
function(tc_case NAME EXPECTED)
    tc_hot_reload_scan(_got ${ARGN})
    if(_got STREQUAL EXPECTED)
        set(_ok 1)
    else()
        set(_ok 0)
    endif()
    set(_code "${_code}    { \"${NAME}\", \"${EXPECTED}\", \"${_got}\", ${_ok} },\n" PARENT_SCOPE)
endfunction()

# tc_decide_case(<name> <expected ON|OFF> <platform-supported> <files>...)
# Same table, through the full decision (platform gate + scan) both callers use.
function(tc_decide_case NAME EXPECTED PLATFORM_SUPPORTED)
    tc_hot_reload_decide(_got ${PLATFORM_SUPPORTED} ${ARGN})
    if(_got STREQUAL EXPECTED)
        set(_ok 1)
    else()
        set(_ok 0)
    endif()
    set(_code "${_code}    { \"${NAME}\", \"${EXPECTED}\", \"${_got}\", ${_ok} },\n" PARENT_SCOPE)
endfunction()

file(WRITE "${FX}/plain.cpp"       "#include \"tcApp.h\"\nTC_HOT_RELOAD(tcApp)\n")
file(WRITE "${FX}/firstline.cpp"    "TC_HOT_RELOAD(tcApp)\n")
file(WRITE "${FX}/indented.cpp"     "int x;\n    TC_HOT_RELOAD(tcApp)\n")
file(WRITE "${FX}/crlf.cpp"         "int x;\r\nTC_HOT_RELOAD(tcApp)\r\n")
file(WRITE "${FX}/linecomment.cpp"  "// TC_HOT_RELOAD(tcApp)\nint x;\n")
file(WRITE "${FX}/trailing.cpp"     "int x; // TC_HOT_RELOAD(tcApp)\n")
file(WRITE "${FX}/block.cpp"        "/* TC_HOT_RELOAD(tcApp) */\nint x;\n")
file(WRITE "${FX}/none.cpp"         "int main() { return 0; }\n")
file(WRITE "${FX}/header.h"         "TC_HOT_RELOAD(tcApp)\n")

tc_case("macro at line start"          ON  "${FX}/plain.cpp")
tc_case("macro on the first line"      ON  "${FX}/firstline.cpp")
tc_case("indented macro"               ON  "${FX}/indented.cpp")
tc_case("CRLF line endings"            ON  "${FX}/crlf.cpp")
tc_case("// comment line"              OFF "${FX}/linecomment.cpp")
tc_case("trailing // comment"          OFF "${FX}/trailing.cpp")
tc_case("/* block comment */"          OFF "${FX}/block.cpp")
tc_case("no macro"                     OFF "${FX}/none.cpp")
tc_case("headers are not scanned"      OFF "${FX}/header.h")
tc_case("missing file is skipped"      OFF "${FX}/does-not-exist.cpp")
tc_case("any file of several"          ON  "${FX}/none.cpp" "${FX}/plain.cpp")

# The #234 layout: App in a src/ subdirectory. Both scans glob src/ recursively.
file(WRITE "${FX}/proj/src/main.cpp"         "int main() { return 0; }\n")
file(WRITE "${FX}/proj/src/app/tcApp.cpp"    "#include \"tcApp.h\"\nTC_HOT_RELOAD(tcApp)\n")
file(GLOB_RECURSE _proj_cpps "${FX}/proj/src/*.cpp")
tc_case("App in a src/ subdirectory"   ON  ${_proj_cpps})

# The platform gate (#329). trussc_app.cmake bakes TRUE / FALSE into the
# generated check, so these are the values both callers pass.
tc_decide_case("unsupported platform + macro"    OFF FALSE "${FX}/plain.cpp")
tc_decide_case("unsupported platform, no macro"  OFF FALSE "${FX}/none.cpp")
tc_decide_case("supported platform + macro"      ON  TRUE  "${FX}/plain.cpp")
tc_decide_case("supported platform, no macro"    OFF TRUE  "${FX}/none.cpp")
tc_decide_case("supported, macro in src/ subdir" ON  TRUE  ${_proj_cpps})

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.c"
"/* generated by core/tests/hotReloadScan/CMakeLists.txt */
#include <stdio.h>
struct Case { const char* name; const char* expected; const char* got; int ok; };
static const struct Case cases[] = {
${_code}};
int main(void) {
    int fails = 0;
    for (unsigned i = 0; i < sizeof(cases) / sizeof(cases[0]); i++) {
        printf(\"%-40s expected %-3s got %-3s %s\\n\", cases[i].name, cases[i].expected,
               cases[i].got, cases[i].ok ? \"PASS\" : \"FAIL\");
        if (!cases[i].ok) fails++;
    }
    printf(\"\\n%s (%d failure%s)\\n\", fails ? \"FAILED\" : \"OK\", fails, fails == 1 ? \"\" : \"s\");
    return fails ? 1 : 0;
}
")
add_executable(hotReloadScan "${CMAKE_CURRENT_BINARY_DIR}/main.c")
