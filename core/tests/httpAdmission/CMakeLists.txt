cmake_minimum_required(VERSION 3.20)
project(httpAdmission LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(ZLIB QUIET)
add_executable(httpAdmission main.cpp)
target_compile_features(httpAdmission PRIVATE cxx_std_17)
target_link_libraries(httpAdmission PRIVATE Threads::Threads)
if(ZLIB_FOUND)
    target_compile_definitions(httpAdmission PRIVATE CPPHTTPLIB_ZLIB_SUPPORT)
    target_link_libraries(httpAdmission PRIVATE ZLIB::ZLIB)
endif()
