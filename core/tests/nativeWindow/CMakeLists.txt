cmake_minimum_required(VERSION 3.16)
project(nativeWindow CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/tc_platform.cmake")
set(_TC_NATIVE_EGL_DEFAULT OFF)
if(TC_GRAPHICS_BACKEND STREQUAL "GLES3")
    set(_TC_NATIVE_EGL_DEFAULT ON)
endif()
option(TC_NATIVE_EGL "Exercise the X11 EGL backend" ${_TC_NATIVE_EGL_DEFAULT})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(CTest)
    add_test(NAME nativeContracts COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/contracts-test")
    add_test(NAME nativeBackendDefault COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/backend-default-test")
    add_executable(nativeWindow main.cpp)
    target_compile_features(nativeWindow PRIVATE cxx_std_20)
    target_include_directories(nativeWindow PRIVATE ../../include/sokol)
    target_link_libraries(nativeWindow PRIVATE X11 Xcursor Xi)
    target_link_options(nativeWindow PRIVATE -Wl,--wrap=XCreateWindow -Wl,--wrap=XDestroyWindow -Wl,--wrap=XMapWindow -Wl,--wrap=XNextEvent)
    if(TC_NATIVE_EGL)
        target_compile_definitions(nativeWindow PRIVATE SOKOL_GLES3)
        target_link_libraries(nativeWindow PRIVATE EGL GLESv2)
        target_link_options(nativeWindow PRIVATE -Wl,--wrap=eglCreateWindowSurface -Wl,--wrap=eglSwapBuffers)
    else()
        target_compile_definitions(nativeWindow PRIVATE SOKOL_GLCORE)
        target_link_libraries(nativeWindow PRIVATE GL)
        target_link_options(nativeWindow PRIVATE -Wl,--wrap=glXCreateWindow -Wl,--wrap=glXSwapBuffers)
    endif()
elseif(WIN32)
    add_executable(nativeWindow win32.cpp)
    target_compile_features(nativeWindow PRIVATE cxx_std_20)
    target_compile_definitions(nativeWindow PRIVATE SOKOL_D3D11)
    target_include_directories(nativeWindow PRIVATE ../../include/sokol)
    target_link_libraries(nativeWindow PRIVATE d3d11 dxgi user32 shell32 gdi32)
    if(MSVC)
        target_compile_options(nativeWindow PRIVATE /utf-8)
    endif()
else()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/skip.cpp" "#include <cstdio>\nint main(){std::puts(\"SKIP: nativeWindow requires X11\");}\n")
    add_executable(nativeWindow "${CMAKE_CURRENT_BINARY_DIR}/skip.cpp")
endif()
