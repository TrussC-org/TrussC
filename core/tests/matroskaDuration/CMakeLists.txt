cmake_minimum_required(VERSION 3.16)
project(matroskaDuration CXX)
add_executable(matroskaDuration main.cpp)
target_compile_features(matroskaDuration PRIVATE cxx_std_17)
target_include_directories(matroskaDuration PRIVATE ../../include)
if(MSVC)
    target_compile_options(matroskaDuration PRIVATE /W4 /utf-8)
else()
    target_compile_options(matroskaDuration PRIVATE -Wall -Wextra -Werror)
endif()
