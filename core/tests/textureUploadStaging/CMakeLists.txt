cmake_minimum_required(VERSION 3.16)
project(textureUploadStaging CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(textureUploadStaging main.cpp)
target_include_directories(textureUploadStaging PRIVATE ../../include/sokol ../../include)
if(MSVC)
    target_compile_options(textureUploadStaging PRIVATE /utf-8)
else()
    target_compile_options(textureUploadStaging PRIVATE -Wall -Wextra -Wno-unused-function)
endif()
