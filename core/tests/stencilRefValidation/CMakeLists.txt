cmake_minimum_required(VERSION 3.16)
project(stencilRefValidation C)
set(CMAKE_C_STANDARD 11)
foreach(mode IN ITEMS debug release)
    set(target stencilRefValidation)
    if(mode STREQUAL "release")
        set(target stencilRefValidation_release)
    endif()
    add_executable(${target} main.c)
    target_include_directories(${target} PRIVATE ../../include/sokol)
    if(mode STREQUAL "debug")
        target_compile_definitions(${target} PRIVATE SOKOL_DEBUG SOKOL_VALIDATE_NON_FATAL)
    else()
        target_compile_definitions(${target} PRIVATE NDEBUG)
    endif()
endforeach()
