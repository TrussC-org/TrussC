cmake_minimum_required(VERSION 3.20)
project(windowRequestNative LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This optional native regression harness requires Linux/X11")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(TC_TEST_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${TC_TEST_CORE}" "${TC_TEST_CORE}/build-linux")
find_package(X11 REQUIRED)
find_library(TC_TEST_XTST Xtst REQUIRED)
add_executable(windowRequestNative main.cpp)
target_link_libraries(windowRequestNative PRIVATE TrussC X11::X11 ${TC_TEST_XTST})
