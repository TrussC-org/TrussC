# Standalone headless test; no TrussC runtime or display required.
cmake_minimum_required(VERSION 3.16)
project(videoNV12Copy CXX)
add_executable(videoNV12Copy main.cpp)
target_compile_features(videoNV12Copy PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(videoNV12Copy PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(videoNV12Copy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
    target_link_libraries(videoNV12Copy PRIVATE PkgConfig::FFMPEG)
    target_compile_definitions(videoNV12Copy PRIVATE TC_TEST_FFMPEG)
endif()
enable_testing()
add_test(NAME videoNV12Copy COMMAND videoNV12Copy)
