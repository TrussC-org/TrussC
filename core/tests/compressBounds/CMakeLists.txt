cmake_minimum_required(VERSION 3.16)
project(compressBounds LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(compressBounds main.cpp ../../include/tc/utils/tcCompress.cpp ../../include/lz4/lz4.c)
target_include_directories(compressBounds PRIVATE mocks ../../include)
if(MSVC)
    target_compile_options(compressBounds PRIVATE /utf-8)
endif()
if(EMSCRIPTEN)
    target_compile_options(compressBounds PRIVATE -fexceptions)
    target_link_options(compressBounds PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1 -sABORTING_MALLOC=0)
endif()
