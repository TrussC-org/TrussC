cmake_minimum_required(VERSION 3.16)
project(sglStencilRef C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(sglStencilRef main.c)
target_include_directories(sglStencilRef PRIVATE ../../include/sokol)
if(MSVC)
    target_compile_options(sglStencilRef PRIVATE /utf-8 /UNDEBUG)
else()
    target_compile_options(sglStencilRef PRIVATE -Wall -Wextra -Wno-unused-function -UNDEBUG)
    target_link_libraries(sglStencilRef PRIVATE m)
endif()
