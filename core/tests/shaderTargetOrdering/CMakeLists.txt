cmake_minimum_required(VERSION 3.20)
project(shaderTargetOrdering C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(TRUSSC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${TRUSSC_DIR}/cmake/trussc_shaders.cmake")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/empty")
trussc_compile_shaders(empty "${CMAKE_CURRENT_BINARY_DIR}/empty")
get_target_property(kind empty_shaders TYPE)
if(NOT kind STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "Missing shader-less interface")
endif()
add_subdirectory(producer)
add_subdirectory(consumer)
