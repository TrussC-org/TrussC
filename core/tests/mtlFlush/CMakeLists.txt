# Standalone sokol test; never link the platform implementation in libTrussC.
cmake_minimum_required(VERSION 3.16)
project(mtlFlush CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(mtlFlush main.cpp)
target_include_directories(mtlFlush PRIVATE ../../include/sokol)
if(APPLE)
    enable_language(OBJCXX)
    set_property(TARGET mtlFlush PROPERTY OBJCXX_STANDARD 20)
    set_source_files_properties(main.cpp PROPERTIES LANGUAGE OBJCXX)
    target_compile_definitions(mtlFlush PRIVATE SOKOL_METAL)
    target_compile_options(mtlFlush PRIVATE -fobjc-arc)
    target_link_libraries(mtlFlush PRIVATE "-framework Metal" "-framework QuartzCore" "-framework Foundation")
else()
    target_compile_definitions(mtlFlush PRIVATE SOKOL_DUMMY_BACKEND)
endif()
if(NOT MSVC)
    target_compile_options(mtlFlush PRIVATE -Wall -Wextra -Wno-unused-function)
endif()
