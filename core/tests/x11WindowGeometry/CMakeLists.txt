cmake_minimum_required(VERSION 3.16)
project(x11WindowGeometry CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(x11WindowGeometry main.cpp)
if(UNIX AND NOT APPLE AND NOT EMSCRIPTEN AND NOT ANDROID)
    find_package(X11 REQUIRED)
    target_include_directories(x11WindowGeometry PRIVATE ${X11_INCLUDE_DIR})
    target_compile_definitions(x11WindowGeometry PRIVATE TC_TEST_X11)
endif()
