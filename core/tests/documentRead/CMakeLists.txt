cmake_minimum_required(VERSION 3.20)
project(documentRead LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(documentRead main.cpp)
target_include_directories(documentRead PRIVATE ../../include ../../../addons/tcxLut/src)
if(MSVC)
    target_compile_options(documentRead PRIVATE /W4)
else()
    target_compile_options(documentRead PRIVATE -Wall -Wextra -Wpedantic)
endif()
