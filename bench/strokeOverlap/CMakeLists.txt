cmake_minimum_required(VERSION 3.20)
project(strokeOverlap)
if(NOT TRUSSC_DIR)
    get_filename_component(TRUSSC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../core" ABSOLUTE)
endif()
option(STROKE_OVERLAP_CLIPPER2 "Enable Clipper2 offset and earcut modes" OFF)
set(CLIPPER2_SOURCE_DIR "" CACHE PATH "Pinned local Clipper2 checkout (never fetched)")
include("${TRUSSC_DIR}/cmake/trussc_app.cmake")
trussc_app(NAME strokeOverlap)
if(STROKE_OVERLAP_CLIPPER2)
    set(_clipper_lib "${CLIPPER2_SOURCE_DIR}/CPP/Clipper2Lib")
    if(NOT EXISTS "${_clipper_lib}/include/clipper2/clipper.h")
        message(FATAL_ERROR "Supply the local pinned Clipper2_1.5.4 source directory")
    endif()
    file(GLOB _clipper_sources CONFIGURE_DEPENDS "${_clipper_lib}/src/*.cpp")
    add_library(strokeOverlapClipper2 STATIC ${_clipper_sources})
    target_include_directories(strokeOverlapClipper2 PUBLIC "${_clipper_lib}/include")
    target_compile_features(strokeOverlapClipper2 PUBLIC cxx_std_17)
    target_link_libraries(strokeOverlap PRIVATE strokeOverlapClipper2)
    target_compile_definitions(strokeOverlap PRIVATE STROKE_OVERLAP_CLIPPER2=1)
endif()
