cmake_minimum_required(VERSION 3.20)
project(snapshotCost LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(APPLE)
  enable_language(OBJC OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "" FORCE)
endif()
option(PROBE_GLES3 "Force desktop Linux to use the Pi GLES3 backend" OFF)
option(PROBE_GLCORE "Force Pi to use desktop GL/GLX (requires driver support)" OFF)
get_filename_component(TRUSSC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../core" ABSOLUTE)
add_subdirectory("${TRUSSC_DIR}" "${CMAKE_BINARY_DIR}/core")
if(UNIX AND NOT APPLE AND (PROBE_GLES3 OR PROBE_GLCORE))
  get_target_property(defs TrussC COMPILE_DEFINITIONS)
  list(REMOVE_ITEM defs SOKOL_GLCORE SOKOL_GLES3)
  if(PROBE_GLES3)
    list(APPEND defs SOKOL_GLES3)
    target_link_libraries(TrussC PUBLIC EGL GLESv2)
  else()
    list(APPEND defs SOKOL_GLCORE)
    find_package(OpenGL REQUIRED)
    target_link_libraries(TrussC PUBLIC OpenGL::GL)
  endif()
  set_property(TARGET TrussC PROPERTY COMPILE_DEFINITIONS "${defs}")
  set_property(TARGET TrussC PROPERTY INTERFACE_COMPILE_DEFINITIONS "${defs}")
endif()
# The executable supplies every Sokol symbol; the stock sokol_impl archive
# member is not extracted. Only this probe TU exposes the pre-commit seam.
add_executable(snapshotCost src/main.cpp src/native.cpp)
if(APPLE)
  set_source_files_properties(src/native.cpp PROPERTIES LANGUAGE OBJCXX)
  target_compile_options(snapshotCost PRIVATE -fobjc-arc)
endif()
target_link_libraries(snapshotCost PRIVATE tc::TrussC)
set_target_properties(snapshotCost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
