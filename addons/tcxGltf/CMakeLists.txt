# =============================================================================
# tcxGltf - TrussC glTF 2.0 Loader Addon
# =============================================================================

cmake_minimum_required(VERSION 3.16)

set(ADDON_NAME tcxGltf)

# Fetch cgltf (single-header C glTF parser)
include(FetchContent)
FetchContent_Declare(
    cgltf
    GIT_REPOSITORY https://github.com/jkuhlmann/cgltf.git
    GIT_TAG v1.15
)
FetchContent_MakeAvailable(cgltf)

# Keep the fetched source intact; C1 adds two small parsing-allocation fixes.
# Verify LF-normalized upstream and generated bytes; see PROVENANCE.md.
file(READ "${cgltf_SOURCE_DIR}/cgltf.h" CGLTF_SOURCE)
string(REPLACE "\r\n" "\n" CGLTF_SOURCE "${CGLTF_SOURCE}")
string(SHA256 CGLTF_SHA256 "${CGLTF_SOURCE}")
if(NOT CGLTF_SHA256 STREQUAL "e378a21c084bf1f288bb799de827bb26906efb024255f1ecf1705ea13f11c6ec")
    message(FATAL_ERROR "cgltf v1.15 upstream SHA-256 mismatch; review PROVENANCE.md before updating C1")
endif()

# Apply exactly the reviewed insertions to the build copy.
set(CGLTF_ATTRIBUTES_FAILURE "\tif (!*out_attributes)\n\t{\n")
string(REPLACE "${CGLTF_ATTRIBUTES_FAILURE}"
    "${CGLTF_ATTRIBUTES_FAILURE}\t\t*out_attributes_count = 0;\n"
    CGLTF_SOURCE "${CGLTF_SOURCE}")
set(CGLTF_MAPPINGS_ALLOCATION "\t\t\tout_prim->mappings = (cgltf_material_mapping*)cgltf_calloc(options, sizeof(cgltf_material_mapping), out_prim->mappings_count);\n")
string(REPLACE "${CGLTF_MAPPINGS_ALLOCATION}"
    "${CGLTF_MAPPINGS_ALLOCATION}\t\t\tif (!out_prim->mappings)\n\t\t\t{\n\t\t\t\tout_prim->mappings_count = 0;\n\t\t\t\treturn CGLTF_ERROR_NOMEM;\n\t\t\t}\n"
    CGLTF_SOURCE "${CGLTF_SOURCE}")
string(SHA256 CGLTF_COPY_SHA256 "${CGLTF_SOURCE}")
if(NOT CGLTF_COPY_SHA256 STREQUAL "ad63ae638f790df8346a2fc7abc56c1a03b46a27e1c1f71a9093999892cebf74")
    message(FATAL_ERROR "cgltf C1 generated-copy SHA-256 mismatch; review the two insertions in PROVENANCE.md")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/cgltf-safe")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cgltf-safe/source.tmp" "${CGLTF_SOURCE}")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/cgltf-safe/source.tmp"
    "${CMAKE_CURRENT_BINARY_DIR}/cgltf-safe/tcxGltfCgltf.h" COPYONLY)
file(REMOVE "${CMAKE_CURRENT_BINARY_DIR}/cgltf-safe/source.tmp")

# Source files
file(GLOB ADDON_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
file(GLOB ADDON_HEADERS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.h)

add_library(${ADDON_NAME} STATIC ${ADDON_SOURCES} ${ADDON_HEADERS})

target_include_directories(${ADDON_NAME} PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# cgltf is header-only; just add include path
target_include_directories(${ADDON_NAME} PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}/cgltf-safe
)

target_link_libraries(${ADDON_NAME} PUBLIC TrussC)

# STL allocation failures must reach the loader's exception boundary on Web too.
if(EMSCRIPTEN)
    target_compile_options(${ADDON_NAME} PRIVATE -fexceptions)
    target_link_options(${ADDON_NAME} INTERFACE -fexceptions -sABORTING_MALLOC=0)
endif()
